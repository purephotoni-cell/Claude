<!-- Pure Photo N.I | Studio Ninja enquiry forms, works with Ajax page loading -->
<script>
(function(){
  var SRC='https://app.studioninja.co/client-assets/form-render/assets/scripts/iframeResizer.js';
  function withResizer(cb){
    if(window.iFrameResize)return cb();
    var s=document.querySelector('script[data-pn-resizer]');
    if(!s){s=document.createElement('script');s.src=SRC;s.setAttribute('data-pn-resizer','1');document.head.appendChild(s)}
    s.addEventListener('load',cb);
  }
  function setup(f){
    if(f.getAttribute('data-pn'))return;
    f.setAttribute('data-pn','1');
    var ok=false,tries=0;
    function init(){if(window.iFrameResize&&!f.iFrameResizer){try{window.iFrameResize({checkOrigin:false,onResized:function(){ok=true}},f)}catch(e){}}}
    function check(){if(ok||tries>1||!document.body.contains(f))return;tries++;f.src=f.getAttribute('src');setTimeout(check,5000)}
    f.addEventListener('load',function(){if(f.iFrameResizer)f.iFrameResizer.resize()});
    withResizer(init);
    setTimeout(check,5000);
  }
  function run(){document.querySelectorAll('iframe[src*="studioninja.co/contactform"]').forEach(setup)}
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',run);else run();
  window.addEventListener('mercury:load',run);
})();
</script>
