(function(){
var buttons=document.querySelectorAll('.pn-v3-filters button');
var cards=document.querySelectorAll('.pn-v3-card');
var count=document.querySelector('.pn-v3-count');
function show(filter){
var shown=0;
for(var i=0;i<cards.length;i++){
var match=filter==='all'||cards[i].getAttribute('data-region')===filter;
cards[i].hidden=!match;
if(match)shown++;
}
for(var j=0;j<buttons.length;j++){
var on=buttons[j].getAttribute('data-filter')===filter;
buttons[j].classList.toggle('is-active',on);
buttons[j].setAttribute('aria-pressed',on?'true':'false');
}
count.textContent='Showing '+shown+(shown===1?' venue':' venues');
}
for(var k=0;k<buttons.length;k++){
buttons[k].addEventListener('click',function(){show(this.getAttribute('data-filter'))});
}
show('all');
})();
</script>
